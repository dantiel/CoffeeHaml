Write This

.card
  %h2= post.title
  - if showBody
    .body= post.content

Get This

import { jsx, jsxs } from "react/jsx-runtime";

jsxs("div", { className: "card" },
  jsx("h2", null, post.title),
  ...(showBody ? [
    jsx("div", { className: "body" }, post.content)
  ] : [])
);
